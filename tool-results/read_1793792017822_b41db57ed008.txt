    11→    11→const clamp01 = (x: number) => Math.max(0, Math.min(1, x));
    12→    12→const clamp = (x: number, a: number, b: number) => Math.max(a, Math.min(b, x));
    13→    13→const easeOutQuint = (p: number) => 1 - Math.pow(1 - p, 5);
    14→    14→const easeInOutQuint = (p: number) =>
    15→    15→  p < 0.5 ? 16 * p * p * p * p * p : 1 - Math.pow(-2 * p + 2, 5) / 2;
    16→    16→const easeOutQuart = (p: number) => 1 - Math.pow(1 - p, 4);

    20→    20→interface ImgEntry {
    21→    21→  img: HTMLImageElement | null;
    22→    22→  ready: boolean;
    23→    23→}
    24→    24→const imageCache = new Map<string, ImgEntry>();

    26→    26→function getImage(src: string): ImgEntry {
    27→    27→  let entry = imageCache.get(src);
    28→    28→  if (!entry) {
    29→    29→    entry = { img: null, ready: false };
    30→    30→    imageCache.set(src, entry);
    31→    31→    if (/^https?:\/\//.test(src) || src.startsWith('/')) {
    32→    32→      const img = new Image();
    33→    33→      img.crossOrigin = 'anonymous';
    34→    34→      img.onload = () => {
    35→    35→        entry.ready = true;
    36→    36→      };
    37→    37→      img.onerror = () => {
    38→    38→        entry.ready = false;
    39→    39→        entry.img = null;
    40→    40→      };
    41→    41→      img.src = src;
    42→    42→      entry.img = img;
    43→    43→    }
    44→    44→  }
    45→    45→  return entry;
    46→    46→}

    55→    55→function pathRoundRect(
    56→    56→  ctx: CanvasRenderingContext2D,
    57→    57→  x: number,
    58→    58→  y: number,
    59→    59→  w: number,
    60→    60→  h: number,
    61→    61→  r: number,
    62→    62→) {
    63→    63→  const rr = Math.min(r, w / 2, h / 2);
    64→    64→  ctx.beginPath();
    65→    65→  ctx.moveTo(x + rr, y);
    66→    66→  ctx.arcTo(x + w, y, x + w, y + h, rr);
    67→    67→  ctx.arcTo(x + w, y + h, x, y + h, rr);
    68→    68→  ctx.arcTo(x, y + h, x, y, rr);
    69→    69→  ctx.arcTo(x, y, x + w, y, rr);
    70→    70→  ctx.closePath();
    71→    71→}

    73→    73→function wrapWords(
    74→    74→  ctx: CanvasRenderingContext2D,
    75→    75→  words: string[],
    76→    76→  maxWidth: number,
    77→    77→): { word: string; w: number }[][] {
    78→    78→  const lines: { word: string; w: number }[][] = [[]];
    79→    79→  let lineW = 0;
    80→    80→  const spaceW = ctx.measureText(' ').width;
    81→    81→  for (const word of words) {
    82→    82→    const w = ctx.measureText(word).width;
    83→    83→    const cur = lines[lines.length - 1];
    84→    84→    if (cur.length > 0 && lineW + spaceW + w > maxWidth) {
    85→    85→      lines.push([{ word, w }]);
    86→    86→      lineW = w;
    87→    87→    } else {
    88→    88→      cur.push({ word, w });
    89→    89→      lineW += (cur.length > 1 ? spaceW : 0) + w;
    90→    90→    }
    91→    91→  }
    92→    92→  return lines;
    93→    93→}

    95→    95→function wrapText(ctx: CanvasRenderingContext2D, text: string, maxWidth: number): string[] {
    96→    96→  const lines: string[] = [];
    97→    97→  for (const para of text.split('\n')) {
    98→    98→    const words = para.split(/\s+/).filter(Boolean);
    99→    99→    if (words.length === 0) {
   100→   100→      lines.push('');
   101→   101→      continue;
   102→   102→    }
   103→   103→    let line = '';
   104→   104→    for (const word of words) {
   105→   105→      const test = line ? `${line} ${word}` : word;
   106→   106→      if (ctx.measureText(test).width > maxWidth && line) {
   107→   107→        lines.push(line);
   108→   108→        line = word;
   109→   109→      } else {
   110→   110→        line = test;
   111→   111→      }
   112→   112→    }
   113→   113→    lines.push(line);
   114→   114→  }
   115→   115→  return lines;
   116→   116→}

   120→   120→function drawAuroraBlobs(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, t: number, pulse: boolean) {
   121→   121→  const { width: W, height: H } = spec;
   122→   122→  const blobs: { c: string; x: number; y: number; r: number; sx: number; sy: number }[] = [
   123→   123→    { c: 'rgba(124, 108, 240, 0.34)', x: 0.24, y: 0.22, r: 0.52, sx: 0.9, sy: 0.6 },
   124→   124→    { c: 'rgba(94, 214, 190, 0.28)', x: 0.82, y: 0.68, r: 0.46, sx: 0.7, sy: 0.9 },
   125→   125→    { c: 'rgba(250, 196, 154, 0.26)', x: 0.62, y: 0.12, r: 0.4, sx: 0.5, sy: 0.8 },
   126→   126→  ];
   127→   127→  const breathe = pulse ? 1 + Math.sin(t / 1600) * 0.06 : 1;
   128→   128→  for (const b of blobs) {
   129→   129→    const bx = (b.x + Math.sin(t / (9000 * b.sx)) * 0.05) * W;
   130→   130→    const by = (b.y + Math.cos(t / (11000 * b.sy)) * 0.04) * H;
   131→   131→    const br = b.r * H * breathe;
   132→   132→    const g = ctx.createRadialGradient(bx, by, 0, bx, by, br);
   133→   133→    g.addColorStop(0, b.c);
   134→   134→    g.addColorStop(1, 'rgba(255,255,255,0)');
   135→   135→    ctx.fillStyle = g;
   136→   136→    ctx.fillRect(0, 0, W, H);
   137→   137→  }
   138→   138→}

   140→   140→function drawBackground(
   141→   141→  ctx: Ctx,
   142→   142→  spec: VideoSpec,
   143→   143→  T: ThemeTokens,
   144→   144→  scene: Scene,
   145→   145→  absT: number,
   146→   146→) {
   147→   147→  const { width: W, height: H } = spec;
   148→   148→  const bg = scene.elements.find((el): el is Extract<SceneElement, { kind: 'bg' }> => el.kind === 'bg');
   149→   149→  const pulse = bg?.anim === 'pulse';
   150→   150→
   151→   151→  if (!bg || bg.variant === 'gradient') {
   152→   152→    const c0 = bg?.colors?.[0] ?? T.bg[0];
   153→   153→    const c1 = bg?.colors?.[1] ?? T.bg[1];
   154→   154→    const g = ctx.createLinearGradient(0, 0, 0, H);
   155→   155→    g.addColorStop(0, c0);
   156→   156→    g.addColorStop(1, c1);
   157→   157→    ctx.fillStyle = g;
   158→   158→    ctx.fillRect(0, 0, W, H);
   159→   159→    if (T.name === 'aurora') drawAuroraBlobs(ctx, spec, T, absT, pulse);
   160→   160→    if (T.name === 'swiss') {
   161→   161→      // faint warm corner tint for depth
   162→   162→      const rg = ctx.createRadialGradient(W * 0.85, H * 0.1, 0, W * 0.85, H * 0.1, H * 0.9);
   163→   163→      rg.addColorStop(0, 'rgba(230, 57, 70, 0.05)');
   164→   164→      rg.addColorStop(1, 'rgba(255,255,255,0)');
   165→   165→      ctx.fillStyle = rg;
   166→   166→      ctx.fillRect(0, 0, W, H);
   167→   167→    }
   168→   168→    return;
   169→   169→  }
   170→   170→
   171→   171→  if (bg.variant === 'solid') {
   172→   172→    ctx.fillStyle = bg.colors[0] ?? T.bg[0];
   173→   173→    ctx.fillRect(0, 0, W, H);
   174→   174→    return;
   175→   175→  }
   176→   176→
   177→   177→  // grid — dotted mesh (aurora) or graph paper (swiss)
   178→   178→  ctx.fillStyle = bg.colors[0] ?? T.bg[0];
   179→   179→  ctx.fillRect(0, 0, W, H);
   180→   180→  ctx.strokeStyle = T.gridLine;
   181→   181→  ctx.fillStyle = T.gridLine;
   182→   182→  ctx.lineWidth = 1;
   183→   183→  if (T.name === 'swiss') {
   184→   184→    const step = W / 16;
   185→   185→    ctx.beginPath();
   186→   186→    for (let x = step; x < W; x += step) {
   187→   187→      ctx.moveTo(x, 0);
   188→   188→      ctx.lineTo(x, H);
   189→   189→    }
   190→   190→    for (let y = step; y < H; y += step) {
   191→   191→      ctx.moveTo(0, y);
   192→   192→      ctx.lineTo(W, y);
   193→   193→    }
   194→   194→    ctx.stroke();
   195→   195→  } else {
   196→   196→    const step = Math.max(40, W / 22);
   197→   197→    for (let x = step / 2; x < W; x += step) {
   198→   198→      for (let y = step / 2; y < H; y += step) {
   199→   199→        ctx.globalAlpha = 0.55;
   200→   200→        ctx.beginPath();
   201→   201→        ctx.arc(x, y, 1.6, 0, Math.PI * 2);
   202→   202→        ctx.fill();
   203→   203→        ctx.globalAlpha = 1;
   204→   204→      }
   205→   205→    }
   206→   206→  }
   207→   207→}

   213→   213→    const seed = i * 127.3;

   229→   229→function titleSize(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, text: string): number {
   230→   230→  const { width: W, height: H } = spec;
   231→   231→  let size = H * 0.1;
   232→   232→  for (let i = 0; i < 8; i++) {
   233→   233→    ctx.font = `800 ${size}px ${T.font}`;
   234→   234→    const lines = wrapText(ctx, text, W * 0.8);
   235→   235→    const fits = lines.every((l) => ctx.measureText(l).width <= W * 0.8);
   236→   236→    if (fits || size < H * 0.04) return size;
   237→   237→    size *= 0.92;
   238→   238→  }
   239→   239→  return size;
   240→   240→}

   242→   242→function drawTitle(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, el: { text: string; anim: string }, elT: number) {
   243→   243→  const { width: W, height: H } = spec;
   244→   244→  const yBase = H * 0.4;
   245→   245→  const size = titleSize(ctx, spec, T, el.text);
   246→   246→
   247→   247→  if (el.anim === 'kinetic') {
   248→   248→    ctx.font = `800 ${size}px ${T.font}`;
   249→   249→    ctx.textBaseline = 'middle';
   250→   250→    ctx.letterSpacing = T.letterSpacing;
   251→   251→    const words = el.text.split(/\s+/).filter(Boolean);
   252→   252→    const lines = wrapWords(ctx, words, W * 0.8);
   253→   253→    const lineHeight = size * 1.2;
   254→   254→    const spaceW = ctx.measureText(' ').width;
   255→   255→    const blockH = lines.length * lineHeight;
   256→   256→    let wi = 0;
   257→   257→    lines.forEach((line, li) => {
   258→   258→      const lineW = line.reduce((s, w) => s + w.w, 0) + spaceW * (line.length - 1);
   259→   259→      let x = (W - lineW) / 2;
   260→   260→      const y = yBase - blockH / 2 + lineHeight * (li + 0.5);
   261→   261→      for (const item of line) {
   262→   262→        const p = easeOutQuint(clamp01((elT - wi * 140) / 700));
   263→   263→        if (p > 0) {
   264→   264→          ctx.save();
   265→   265→          ctx.globalAlpha *= p;
   266→   266→          const dy = (1 - p) * 22;
   267→   267→          ctx.translate(x + item.w / 2, y + dy);
   268→   268→          ctx.scale(0.97 + 0.03 * p, 0.97 + 0.03 * p);
   269→   269→          ctx.fillStyle = T.text;
   270→   270→          ctx.textAlign = 'center';
   271→   271→          ctx.fillText(item.word, 0, 0);
   272→   272→          ctx.restore();
   273→   273→        }
   274→   274→        x += item.w + spaceW;
   275→   275→        wi++;
   276→   276→      }
   277→   277→    });
   278→   278→    ctx.letterSpacing = '0px';
   279→   279→    return;
   280→   280→  }
   281→   281→
   282→   282→  if (el.anim === 'typewriter') {
   283→   283→    ctx.font = `800 ${size}px ${T.font}`;
   284→   284→    ctx.textAlign = 'center';
   285→   285→    ctx.textBaseline = 'middle';
   286→   286→    ctx.letterSpacing = T.letterSpacing;
   287→   287→    const dur = Math.min(Math.max(900, el.text.length * 46), 4200);
   288→   288→    const p = clamp01(elT / dur);
   289→   289→    const shown = el.text.slice(0, Math.ceil(el.text.length * p));
   290→   290→    const fullW = ctx.measureText(el.text).width;
   291→   291→    ctx.save();
   292→   292→    ctx.globalAlpha *= 1;
   293→   293→    if (fullW > W * 0.8) {
   294→   294→      ctx.translate(W / 2, yBase);
   295→   295→      const s = (W * 0.8) / fullW;
   296→   296→      ctx.scale(s, s);
   297→   297→    } else {
   298→   298→      ctx.translate(W / 2, yBase);
   299→   299→    }
   300→   300→    ctx.fillStyle = T.text;
   301→   301→    ctx.fillText(shown, 0, 0);
   302→   302→    const caretA = p < 1 ? 0.85 : Math.sin(elT / 420) > 0 ? 0.8 : 0;
   303→   303→    if (caretA > 0) {
   304→   304→      const cw = ctx.measureText(shown).width;
   305→   305→      ctx.fillStyle = T.palette[1] ?? T.text;
   306→   306→      ctx.globalAlpha *= caretA;
   307→   307→      ctx.fillRect(cw / 2 + 6, -size * 0.42, Math.max(2, size * 0.05), size * 0.84);
   308→   308→    }
   309→   309→    ctx.restore();
   310→   310→    ctx.letterSpacing = '0px';
   311→   311→    return;
   312→   312→  }
   313→   313→
   314→   314→  // pop | fade | fade-up — whole block, ultra smooth
   315→   315→  const p = easeOutQuint(clamp01(elT / ELEM_DUR));
   316→   316→  if (p <= 0) return;
   317→   317→  ctx.font = `800 ${size}px ${T.font}`;
   318→   318→  ctx.textAlign = 'center';
   319→   319→  ctx.textBaseline = 'middle';
   320→   320→  ctx.letterSpacing = T.letterSpacing;
   321→   321→  ctx.fillStyle = T.text;
   322→   322→  const lines = wrapText(ctx, el.text, W * 0.8);
   323→   323→  const lineHeight = size * 1.2;
   324→   324→  const blockH = lines.length * lineHeight;
   325→   325→  ctx.save();
   326→   326→  ctx.globalAlpha *= p;
   327→   327→  ctx.translate(W / 2, yBase - blockH / 2 + lineHeight / 2 + (1 - p) * 18);
   328→   328→  if (el.anim === 'pop') ctx.scale(0.96 + 0.04 * p, 0.96 + 0.04 * p);
   329→   329→  lines.forEach((line, i) => ctx.fillText(line, 0, i * lineHeight));
   330→   330→  ctx.restore();
   331→   331→  ctx.letterSpacing = '0px';
   332→   332→  return;
   333→   333→}

   335→   335→function drawSubtitle(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, el: { text: string; anim: string }, elT: number, hasPie = false) {
   336→   336→  const { width: W, height: H } = spec;
   337→   337→  const y = hasPie ? H * 0.88 : H * 0.56;
   338→   338→  const size = H * 0.05;
   339→   339→  ctx.font = `500 ${size}px ${T.font}`;
   340→   340→  ctx.textAlign = 'center';
   341→   341→  ctx.textBaseline = 'middle';
   342→   342→  ctx.letterSpacing = '1.5px';
   343→   343→
   344→   344→  if (el.anim === 'typewriter') {
   345→   345→    const dur = Math.min(Math.max(800, el.text.length * 40), 3200);
   346→   346→    const p = clamp01(elT / dur);
   347→   347→    const shown = el.text.slice(0, Math.ceil(el.text.length * p));
   348→   348→    ctx.globalAlpha *= 0.95;
   349→   349→    ctx.fillStyle = T.dim;
   350→   350→    ctx.fillText(shown, W / 2, y);
   351→   351→    const cw = ctx.measureText(shown).width;
   352→   352→    ctx.fillStyle = T.palette[1] ?? T.text;
   353→   353→    ctx.globalAlpha *= Math.sin(elT / 380) > 0 ? 0.7 : 0;
   354→   354→    ctx.fillRect(W / 2 + cw / 2 + 8, y - size * 0.4, 2.5, size * 0.8);
   355→   355→    ctx.letterSpacing = '0px';
   356→   356→    return;
   357→   357→  }
   358→   358→
   359→   359→  const p = easeOutQuint(clamp01(elT / ELEM_DUR));
   360→   360→  if (p <= 0) return;
   361→   361→  ctx.save();
   362→   362→  ctx.globalAlpha *= p * 0.95;
   363→   363→  ctx.fillStyle = T.dim;
   364→   364→  const dy = el.anim === 'fade-up' ? (1 - p) * 14 : 0;
   365→   365→  ctx.fillText(el.text, W / 2, y + dy);
   366→   366→  ctx.restore();
   367→   367→  ctx.letterSpacing = '0px';
   368→   368→}

   370→   370→function drawBodyText(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, el: { text: string; anim: string }, elT: number, hasSubtitle: boolean) {
   371→   371→  const { width: W, height: H } = spec;

   373→   373→  const y = (hasSubtitle ? 0.7 : 0.52) * H;
   374→   374→  const p = easeOutQuint(clamp01(elT / ELEM_DUR));
   375→   375→  if (p <= 0) return;
   376→   376→  ctx.font = `400 ${size}px ${T.font}`;
   377→   377→  ctx.textAlign = 'center';
   378→   378→  ctx.textBaseline = 'middle';

   380→   380→  const lineHeight = size * 1.55;
   381→   381→  ctx.save();
   382→   382→  ctx.globalAlpha *= p * 0.92;
   383→   383→  ctx.fillStyle = T.dim;
   384→   384→  lines.forEach((line, i) => ctx.fillText(line, W / 2, y - ((lines.length - 1) * lineHeight) / 2 + i * lineHeight + (1 - p) * 12));
   385→   385→  ctx.restore();
   386→   386→}

   390→   390→function drawBars(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, el: { data: { label: string; value: number }[] }, elT: number) {
   391→   391→  const { width: W, height: H } = spec;
   392→   392→  const data = el.data.slice(0, 8);
   393→   393→  if (data.length === 0) return;
   394→   394→  const maxV = Math.max(...data.map((d) => d.value), 0.0001);
   395→   395→  const hasTitle = true;
   396→   396→  const top = H * (hasTitle ? 0.52 : 0.42);
   397→   397→  const bottom = H * 0.85;
   398→   398→  const margin = W * 0.12;
   399→   399→  const areaW = W - margin * 2;
   400→   400→  const slotW = areaW / data.length;
   401→   401→  const barW = slotW * 0.5;
   402→   402→
   403→   403→  // grid lines
   404→   404→  ctx.strokeStyle = T.gridLine;
   405→   405→  ctx.lineWidth = 1;
   406→   406→  for (let i = 1; i <= 3; i++) {
   407→   407→    const gy = bottom - ((bottom - top) / 3) * i;
   408→   408→    ctx.beginPath();
   409→   409→    ctx.moveTo(margin, gy);
   410→   410→    ctx.lineTo(W - margin, gy);
   411→   411→    ctx.stroke();
   412→   412→  }
   413→   413→  // baseline
   414→   414→  ctx.strokeStyle = T.name === 'swiss' ? 'rgba(17,17,17,0.35)' : 'rgba(22,22,29,0.25)';
   415→   415→  ctx.beginPath();
   416→   416→  ctx.moveTo(margin, bottom);
   417→   417→  ctx.lineTo(W - margin, bottom);
   418→   418→  ctx.stroke();
   419→   419→
   420→   420→  data.forEach((d, i) => {
   421→   421→    const p = easeOutQuart(clamp01((elT - i * 140) / ELEM_DUR));
   422→   422→    if (p <= 0) return;
   423→   423→    const h = (d.value / maxV) * (bottom - top) * p;
   424→   424→    const x = margin + slotW * i + (slotW - barW) / 2;
   425→   425→    const y = bottom - h;
   426→   426→    const color = T.palette[i % T.palette.length];
   427→   427→    ctx.save();
   428→   428→    if (T.glow) {
   429→   429→      ctx.shadowColor = color;
   430→   430→      ctx.shadowBlur = 22;
   431→   431→    } else {
   432→   432→      ctx.shadowColor = 'rgba(22, 22, 29, 0.20)';
   433→   433→      ctx.shadowBlur = 16;
   434→   434→      ctx.shadowOffsetY = 6;
   435→   435→    }
   436→   436→    ctx.fillStyle = color;
   437→   437→    pathRoundRect(ctx, x, y, barW, h, Math.min(12, barW / 2.4));
   438→   438→    ctx.fill();
   439→   439→    ctx.restore();
   440→   440→
   441→   441→    ctx.textAlign = 'center';
   442→   442→    ctx.textBaseline = 'middle';
   443→   443→    ctx.globalAlpha *= p;
   444→   444→    ctx.fillStyle = T.text;
   445→   445→    ctx.font = `700 ${H * 0.032}px ${T.font}`;
   446→   446→    ctx.fillText(String(d.value), x + barW / 2, y - H * 0.03);
   447→   447→    ctx.fillStyle = T.dim;
   448→   448→    ctx.font = `600 ${H * 0.03}px ${T.font}`;
   449→   449→    ctx.fillText(d.label, x + barW / 2, bottom + H * 0.045);
   450→   450→    ctx.globalAlpha = 1;
   451→   451→  });
   452→   452→}

   454→   454→function drawPie(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, el: { data: { label: string; value: number }[] }, elT: number) {
   455→   455→  const { width: W, height: H } = spec;
   456→   456→  const data = el.data.slice(0, 6);
   457→   457→  if (data.length === 0) return;
   458→   458→  const total = data.reduce((s, d) => s + Math.max(0, d.value), 0) || 1;
   459→   459→  const hasLegend = true;
   460→   460→  const cx = hasLegend ? W * 0.36 : W * 0.5;
   461→   461→  const cy = H * 0.55;
   462→   462→  const r = Math.min(H * 0.21, W * 0.18);
   463→   463→  const p = easeInOutQuint(clamp01(elT / 1100));
   464→   464→
   465→   465→  let start = -Math.PI / 2;
   466→   466→  const ringR = r * 0.79;
   467→   467→  const ringW = r * 0.42;
   468→   468→  data.forEach((d, i) => {
   469→   469→    const frac = Math.max(0, d.value) / total;
   470→   470→    const sweep = frac * Math.PI * 2 * p;
   471→   471→    if (sweep <= 0) return;
   472→   472→    const color = T.palette[i % T.palette.length];
   473→   473→    ctx.save();
   474→   474→    if (T.glow) {
   475→   475→      ctx.shadowColor = color;
   476→   476→      ctx.shadowBlur = 18;
   477→   477→    } else {
   478→   478→      ctx.shadowColor = 'rgba(22,22,29,0.18)';
   479→   479→      ctx.shadowBlur = 14;
   480→   480→      ctx.shadowOffsetY = 5;
   481→   481→    }
   482→   482→    ctx.strokeStyle = color;
   483→   483→    ctx.globalAlpha = 0.94;
   484→   484→    ctx.lineWidth = ringW;
   485→   485→    ctx.lineCap = 'butt';
   486→   486→    ctx.beginPath();
   487→   487→    ctx.arc(cx, cy, ringR, start, start + sweep);
   488→   488→    ctx.stroke();
   489→   489→    ctx.restore();
   490→   490→    start += sweep;
   491→   491→  });
   492→   492→
   493→   493→  // legend
   494→   494→  ctx.textAlign = 'left';
   495→   495→  ctx.textBaseline = 'middle';
   496→   496→  data.forEach((d, i) => {
   497→   497→    const lp = easeOutQuint(clamp01((elT - 300 - i * 120) / 600));
   498→   498→    if (lp <= 0) return;
   499→   499→    const color = T.palette[i % T.palette.length];
   500→   500→    const y = H * 0.42 + i * H * 0.085;
   501→   501→    ctx.save();
   502→   502→    ctx.globalAlpha *= lp;
   503→   503→    ctx.fillStyle = color;
   504→   504→    pathRoundRect(ctx, W * 0.6, y - H * 0.016, H * 0.032, H * 0.032, 4);
   505→   505→    ctx.fill();
   506→   506→    ctx.fillStyle = T.text;
   507→   507→    ctx.font = `600 ${H * 0.034}px ${T.font}`;
   508→   508→    ctx.fillText(d.label, W * 0.6 + H * 0.05, y);
   509→   509→    const pct = Math.round((Math.max(0, d.value) / total) * 100);
   510→   510→    ctx.fillStyle = T.dim;
   511→   511→    ctx.font = `500 ${H * 0.03}px ${T.font}`;
   512→   512→    ctx.fillText(`${pct}%`, W * 0.6 + H * 0.05 + ctx.measureText(d.label).width + H * 0.03, y);
   513→   513→    ctx.restore();
   514→   514→  });
   515→   515→}

   519→   519→function drawImage(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, el: { src: string; anim: string }, elT: number, sceneDur: number) {
   520→   520→  const { width: W, height: H } = spec;
   521→   521→  const entry = getImage(el.src);
   522→   522→  const p = easeOutQuint(clamp01(elT / 900));
   523→   523→  if (p <= 0) return;
   524→   524→  const w = W * 0.58;
   525→   525→  const h = H * 0.5;
   526→   526→  const x = (W - w) / 2;
   527→   527→  const y = H * 0.52 - h / 2;
   528→   528→  const radius = H * 0.035;
   529→   529→
   530→   530→  ctx.save();
   531→   531→  ctx.globalAlpha *= p;
   532→   532→
   533→   533→  if (entry.ready && entry.img) {
   534→   534→    const kb = 1 + 0.06 * easeInOutQuint(clamp01(elT / Math.max(1, sceneDur))); // Ken Burns
   535→   535→    const img = entry.img;
   536→   536→    const scale = Math.max(w / img.width, h / img.height) * kb;
   537→   537→    const dw = img.width * scale;
   538→   538→    const dh = img.height * scale;
   539→   539→    if (T.glow) {
   540→   540→      ctx.shadowColor = 'rgba(22,22,29,0.35)';
   541→   541→      ctx.shadowBlur = 30;
   542→   542→      ctx.shadowOffsetY = 12;
   543→   543→    } else {
   544→   544→      ctx.shadowColor = 'rgba(22,22,29,0.22)';
   545→   545→      ctx.shadowBlur = 26;
   546→   546→      ctx.shadowOffsetY = 10;
   547→   547→    }
   548→   548→    ctx.fillStyle = '#ffffff';
   549→   549→    pathRoundRect(ctx, x, y, w, h, radius);
   550→   550→    ctx.fill();
   551→   551→    ctx.shadowColor = 'transparent';
   552→   552→    ctx.save();
   553→   553→    pathRoundRect(ctx, x + 5, y + 5, w - 10, h - 10, radius - 4);
   554→   554→    ctx.clip();
   555→   555→    ctx.drawImage(img, x + (w - dw) / 2, y + (h - dh) / 2, dw, dh);
   556→   556→    ctx.restore();
   557→   557→    ctx.strokeStyle = 'rgba(255,255,255,0.9)';
   558→   558→    ctx.lineWidth = 1.5;
   559→   559→    pathRoundRect(ctx, x, y, w, h, radius);
   560→   560→    ctx.stroke();
   561→   561→  } else {
   562→   562→    // graceful placeholder
   563→   563→    ctx.setLineDash([10, 8]);
   564→   564→    ctx.strokeStyle = T.dim;
   565→   565→    ctx.lineWidth = 2;
   566→   566→    pathRoundRect(ctx, x, y, w, h, radius);
   567→   567→    ctx.stroke();
   568→   568→    ctx.setLineDash([]);
   569→   569→    ctx.fillStyle = T.dim;
   570→   570→    ctx.textAlign = 'center';
   571→   571→    ctx.textBaseline = 'middle';
   572→   572→    ctx.font = `500 ${H * 0.032}px ${T.font}`;
   573→   573→    ctx.fillText('image', W / 2, H * 0.52);
   574→   574→  }
   575→   575→  ctx.restore();
   576→   576→}

   578→   578→function drawShape(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, el: { variant: string; anim: string }, elT: number, absT: number) {
   579→   579→  const { width: W, height: H } = spec;
   580→   580→  const p = easeOutQuint(clamp01(elT / 900));
   581→   581→  if (p <= 0) return;
   582→   582→  ctx.save();
   583→   583→  ctx.globalAlpha *= p;
   584→   584→
   585→   585→  if (el.variant === 'circle') {
   586→   586→    const breathe = el.anim === 'pulse' ? 1 + Math.sin(absT / 1800) * 0.04 : 1;
   587→   587→    const cx = W * 0.5 + Math.sin(absT / 2600) * W * 0.015;
   588→   588→    const cy = H * 0.42;
   589→   589→    const r = H * 0.3 * breathe;
   590→   590→    const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r);
   591→   591→    if (T.name === 'swiss') {
   592→   592→      g.addColorStop(0, 'rgba(230, 57, 70, 0.10)');
   593→   593→      g.addColorStop(1, 'rgba(230, 57, 70, 0)');
   594→   594→    } else {
   595→   595→      g.addColorStop(0, 'rgba(255, 255, 255, 0.65)');
   596→   596→      g.addColorStop(1, 'rgba(255, 255, 255, 0.08)');
   597→   597→    }
   598→   598→    ctx.fillStyle = g;
   599→   599→    ctx.beginPath();
   600→   600→    ctx.arc(cx, cy, r, 0, Math.PI * 2);
   601→   601→    ctx.fill();
   602→   602→    if (T.name === 'aurora') {
   603→   603→      ctx.strokeStyle = 'rgba(255,255,255,0.8)';
   604→   604→      ctx.lineWidth = 1.5;
   605→   605→      ctx.stroke();
   606→   606→    }
   607→   607→  } else if (el.variant === 'ring') {
   608→   608→    ctx.strokeStyle = T.name === 'swiss' ? 'rgba(17,17,17,0.3)' : 'rgba(124,108,240,0.4)';
   609→   609→    ctx.lineWidth = 2.5;
   610→   610→    ctx.beginPath();
   611→   611→    ctx.arc(W / 2, H * 0.44, H * 0.26, 0, Math.PI * 2);
   612→   612→    ctx.stroke();
   613→   613→  } else {
   614→   614→    // glass frame
   615→   615→    const inset = W * 0.05;
   616→   616→    ctx.strokeStyle = T.name === 'swiss' ? 'rgba(17,17,17,0.25)' : 'rgba(255,255,255,0.85)';
   617→   617→    ctx.fillStyle = T.name === 'swiss' ? 'rgba(255,255,255,0)' : 'rgba(255,255,255,0.30)';
   618→   618→    ctx.lineWidth = 2;
   619→   619→    pathRoundRect(ctx, inset, H * 0.08, W - inset * 2, H * 0.84, H * 0.04);
   620→   620→    ctx.fill();
   621→   621→    ctx.stroke();
   622→   622→  }
   623→   623→  ctx.restore();
   624→   624→}

   628→   628→function drawSceneElements(

   651→   651→        break;
   652→   652→      case 'bars':
   653→   653→        drawBars(ctx, spec, T, el, elT);
   654→   654→        break;
   655→   655→      case 'pie':
   656→   656→        drawPie(ctx, spec, T, el, elT);
   657→   657→        break;
   658→   658→      case 'image':
   659→   659→        drawImage(ctx, spec, T, el, elT, scene.durationMs);
   660→   660→        break;
   661→   661→      case 'shape':
   662→   662→        drawShape(ctx, spec, T, el, elT, absT);
   663→   663→        break;

   668→   668→  }

   671→   671→function drawTitleBar(ctx: Ctx, spec: VideoSpec, T: ThemeTokens, scene: Scene, localT: number) {
   672→   672→  if (!T.titleBar) return;

   674→   674→  if (!title) return;

   676→   676→  if (p <= 0) return;
   677→   677→  const { width: W, height: H } = spec;

   679→   679→  const w = W * 0.09 * p;
   680→   680→  ctx.save();
   681→   681→  ctx.globalAlpha *= 0.9;
   682→   682→  ctx.fillStyle = T.palette[1];
   683→   683→  pathRoundRect(ctx, W / 2 - w / 2, y, w, Math.max(4, H * 0.008), 99);

   688→   688→function drawScene(
   689→   689→  ctx: Ctx,
   690→   690→  spec: VideoSpec,
   691→   691→  scene: Scene,
   692→   692→  localT: number,
   693→   693→  alphaMul: number,
   694→   694→  settled: boolean,
   695→   695→  T: ThemeTokens,
   696→   696→) {
   697→   697→  const absT = scene.startMs + localT;
   698→   698→  const p = settled ? 1 : easeInOutQuint(clamp01(localT / TRANSITION_MS));
   699→   699→  const { width: W, height: H } = spec;
   700→   700→
   701→   701→  ctx.save();
   702→   702→  ctx.globalAlpha = alphaMul;
   703→   703→
   704→   704→  // scene enter motion — swift and smooth
   705→   705→  if (!settled && p < 1) {
   706→   706→    switch (scene.enter) {
   707→   707→      case 'fade':
   708→   708→        ctx.globalAlpha = alphaMul * p;
   709→   709→        break;
   710→   710→      case 'fade-up':
   711→   711→        ctx.globalAlpha = alphaMul * p;
   712→   712→        ctx.translate(0, (1 - p) * H * 0.05);
   713→   713→        break;
   714→   714→      case 'slide-left':
   715→   715→        ctx.globalAlpha = alphaMul * Math.min(1, p * 1.4);
   716→   716→        ctx.translate((1 - p) * W * 0.12, 0);
   717→   717→        break;
   718→   718→      case 'slide-right':
   719→   719→        ctx.globalAlpha = alphaMul * Math.min(1, p * 1.4);
   720→   720→        ctx.translate(-(1 - p) * W * 0.12, 0);
   721→   721→        break;
   722→   722→      case 'zoom-in':
   723→   723→        ctx.globalAlpha = alphaMul * p;
   724→   724→        ctx.translate(W / 2, H / 2);
   725→   725→        ctx.scale(0.965 + 0.035 * p, 0.965 + 0.035 * p);
   726→   726→        ctx.translate(-W / 2, -H / 2);
   727→   727→        break;
   728→   728→      default:
   729→   729→        break;
   730→   730→    }
   731→   731→  }
   732→   732→
   733→   733→  drawBackground(ctx, spec, T, scene, absT);
   734→   734→  if (T.particles > 0) drawBokeh(ctx, spec, absT);
   735→   735→  drawSceneElements(ctx, spec, T, scene, localT, absT);
   736→   736→  drawTitleBar(ctx, spec, T, scene, localT);
   737→   737→
   738→   738→  ctx.restore();
   739→   739→}

   768→   768→/** Render one frame at absolute time tMs. */
   769→   769→export function renderFrame(ctx: Ctx, spec: VideoSpec, tMs: number) {
   770→   770→  const T = getTheme(spec.theme);
   771→   771→  const { width: W, height: H } = spec;
   772→   772→  const t = clamp(tMs, 0, Math.max(1, spec.totalMs));
   773→   773→
   774→   774→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   775→   775→  ctx.globalAlpha = 1;
   776→   776→  ctx.shadowBlur = 0;
   777→   777→  ctx.shadowColor = 'transparent';
   778→   778→  ctx.fillStyle = T.bg[0];
   779→   779→  ctx.fillRect(0, 0, W, H);
   780→   780→
   781→   781→  if (spec.scenes.length === 0) {
   782→   782→    ctx.fillStyle = T.dim;
   783→   783→    ctx.textAlign = 'center';
   784→   784→    ctx.textBaseline = 'middle';
   785→   785→    ctx.font = `500 ${H * 0.045}px ${T.font}`;
   786→   786→    ctx.fillText('Write your first scene in the editor', W / 2, H / 2);
   787→   787→    return;
   788→   788→  }
   789→   789→
   790→   790→  const idx = activeSceneIndex(spec, t);
   791→   791→  const scene = spec.scenes[idx];
   792→   792→  const localT = t - scene.startMs;
   793→   793→
   794→   794→  if (idx > 0 && localT < TRANSITION_MS) {
   795→   795→    const prev = spec.scenes[idx - 1];
   796→   796→    drawScene(ctx, spec, prev, prev.durationMs, 1 - easeInOutQuint(clamp01(localT / TRANSITION_MS)), true, T);
   797→   797→  }
   798→   798→  drawScene(ctx, spec, scene, localT, 1, false, T);
   799→   799→  drawOverlays(ctx, spec, T);
   800→   800→}

   802→   802→/** Time when every entrance animation in the scene has settled (for slideshow frames). */
   803→   803→export function sceneSettleMs(scene: Scene): number {
   804→   804→  const content = scene.elements.filter((el) => el.kind !== 'bg');
   805→   805→  let maxEnd = 900;
   806→   806→  content.forEach((el, i) => {
   807→   807→    const delay = ELEM_DELAY + i * ELEM_STAGGER;
   808→   808→    let end = delay + ELEM_DUR;
   809→   809→    if (el.kind === 'title' && el.anim === 'kinetic') {
   810→   810→      end = delay + el.text.split(/\s+/).length * 140 + 700;
   811→   811→    } else if ((el.kind === 'title' || el.kind === 'subtitle') && el.anim === 'typewriter') {
   812→   812→      end = delay + Math.min(Math.max(900, el.text.length * 46), 4200);
   813→   813→    } else if (el.kind === 'pie') {
   814→   814→      end = delay + 1100 + el.data.length * 120;
   815→   815→    }
   816→   816→    maxEnd = Math.max(maxEnd, end);
   817→   817→  });
   818→   818→  return Math.min(Math.max(maxEnd + 300, 900), Math.max(900, scene.durationMs - 60));
   819→   819→}
